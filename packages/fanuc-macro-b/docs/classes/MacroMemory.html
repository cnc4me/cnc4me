        An unsubscribe method.

        Example

        import Emittery from 'emittery';

        const emitter = new Emittery();

        emitter.on('🦄', data => {
        	console.log(data);
        });

        emitter.on(['🦄', '🐶'], data => {
        	console.log(data);
        });

        emitter.emit('🦄', '🌈'); // log => '🌈' x2
        emitter.emit('🐶', '🍖'); // log => '🍖'
